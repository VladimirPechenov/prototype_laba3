2.1.  Что  такое  полировка  
Полировка  (polish)  –  это  доведение  игры  до  состояния,  когда  она  ощущается

public class Checkpoint : MonoBehaviour {     private static Vector3 lastCheckpointPos;     private static int lastCheckpointHealth;     private static int lastCheckpointCoins;      void OnTriggerEnter(Collider other)     {         if (other.CompareTag("Player"))         {             lastCheckpointPos  = other.transform.position;             PlayerHealth health  = other.GetComponent<PlayerHealth>();             lastCheckpointHealth  = health.currentHealth;             ResourceManager rm  = FindObjectOfType<ResourceManager>();             lastCheckpointCoins  = rm.coins;             //  визуальный  эффект         }

    }      public static void RespawnPlayer(GameObject player)     {         player.transform.position  = lastCheckpointPos;         PlayerHealth health  = player.GetComponent<PlayerHealth>();         health.currentHealth  = lastCheckpointHealth;         health.UpdateUI();         ResourceManager.Instance.coins  = lastCheckpointCoins;         ResourceManager.Instance.UpdateUI();     }

    {         if (pool.Count  > 0)         {             GameObject obj  = pool.Dequeue();             obj.SetActive(true);             return obj;         }         else         {             return Instantiate(bulletPrefab);         }     }      public void ReturnBullet(GameObject obj)     {         obj.SetActive(false);         pool.Enqueue(obj);     }
